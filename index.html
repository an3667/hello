=== Màn hình === CHIEU_RONG = 950 CHIEU_CAO = 600 man_hanh = pygame.display.set_mode((CHIEU_RONG, CHIEU_CAO)) pygame.display.set_caption("🔑 CHIIKAWA — Tìm Chìa Khóa | Giống OibKing 🔊") # === Màu sắc === TRANG = (255, 255, 255) XANH_NHE = (204, 255, 255) VANG = (255, 215, 0) VANG_NHE = (255, 255, 204) HONG_NHE = (255, 229, 242) XANH_LA = (153, 255, 153) XANH_DAM = (51, 153, 51) NAU = (139, 69, 19) DO = (255, 80, 80) XAM = (150, 150, 150) # ================================================== # HỆ THỐNG ÂM THANH — TỰ TẠO, KHÔNG CẦN FILE NGOÀI # ================================================== def tao_tieng(dau, cuoi, thoi_gian=0.15, am_luong=0.3): tan_so = 44100 do_dai = int(tan_so * thoi_gian) du_lieu = bytearray() for i in range(do_dai): t = i / tan_so ti_le = i / do_dai f = dau + (cuoi - dau) * ti_le giam_am = 1 - ti_le ** 1.5 gia_tri = int(math.sin(2 * math.pi * f * t) * 32767 * am_luong * giam_am) du_lieu += gia_tri.to_bytes(2, 'little', signed=True) du_lieu += gia_tri.to_bytes(2, 'little', signed=True) return pygame.mixer.Sound(buffer=du_lieu) # Tạo tất cả hiệu ứng âm thanh tieng_nhay = tao_tieng(420, 880, 0.12, 0.25) tieng_lay_chia_khoa = tao_tieng(523, 1047, 0.2, 0.3) tieng_mo_cua = tao_tieng(350, 720, 0.28, 0.3) tieng_qua_muc = tao_tieng(500, 1000, 0.3, 0.3) tieng_thua = tao_tieng(420, 110, 0.45, 0.35) # Mắt nhắm — giống hình pygame.draw.arc(man_hanh, (0,0,0), (int(x+11*k), int(y+14*k), int(11*k), int(6*k)), 3.14, 6.28, 2) pygame.draw.arc(man_hanh, (0,0,0), (int(x+30*k), int(y+14*k), int(11*k), int(6*k)), 3.14, 6.28, 2) # Má hồng pygame.draw.circle(man_hanh, (255,153,153), (int(x+9*k), int(y+27*k)), int(5*k)) pygame.draw.circle(man_hanh, (255,153,153), (int(x+43*k), int(y+27*k)), int(5*k)) # Miệng cười pygame.draw.arc(man_hanh, (100,100,100), (int(x+18*k), int(y+32*k), int(16*k), int(9*k)), 3.14, 6.28, 2) pygame.draw.line(man_hanh, TRANG, (int(x+26*k), int(y+38*k)), (int(x+30*k), int(y+38*k)), 2) # === Vẽ chìa khóa vàng === def ve_chia_khoa(vt): x, y = vt pygame.draw.circle(man_hanh, VANG, (x+12, y+12), 10, 3) pygame.draw.rect(man_hanh, VANG, (x+9, y+20, 5, 20)) pygame.draw.rect(man_hanh, VANG, (x+11, y+32, 6, 4)) pygame.draw.rect(man_hanh, VANG, (x+11, y+26, 4, 4)) # Ánh sáng pygame.draw.circle(man_hanh, TRANG, (x+8, y+8), 2) # === Vẽ quái vật === def ve_quai(vt): x, y = vt pygame.draw.ellipse(man_hanh, DO, (x, y+10, 45, 40)) pygame.draw.circle(man_hanh, DO, (x+22, y+14), 21) # Mắt trắng đen pygame.draw.circle(man_hanh, TRANG, (x+12, y+11), 7) pygame.draw.circle(man_hanh, TRANG, (x+32, y+11), 7) pygame.draw.circle(man_hanh, (0,0,0), (x+13, y+12), 3) pygame.draw.circle(man_hanh, (0,0,0), (x+33, y+12), 3) # Răng pygame.draw.polygon(man_hanh, (60,0,0), [(x+12, y+30), (x+18, y+36), (x+24, y+30)]) pygame.draw.polygon(man_hanh, (60,0,0), [(x+24, y+30), (x+30, y+36), (x+36, y+30)]) # === Vẽ cửa ra === def ve_cua(da_mo): if not da_mo: pygame.draw.rect(man_hanh, NAU, cua_ra) pygame.draw.rect(man_hanh, VANG, (cua_ra.x+7, cua_ra.y+7, cua_ra.w-14, cua_ra.h-14), 4) pygame.draw.circle(man_hanh, VANG, (cua_ra.x+cua_ra.w-16, cua_ra.y+cua_ra.h//2), 7) pygame.draw.rect(man_hanh, XAM, (cua_ra.x+18, cua_ra.y+cua_ra.h//2-6, 18, 12)) else: pygame.draw.rect(man_hanh, XANH_DAM, cua_ra) pygame.draw.polygon(man_hanh, VANG, [(cua_ra.x+20, cua_ra.y+28), (cua_ra.x+36, cua_ra.y+16), (cua_ra.x+52, cua_ra.y+38)]) # === Bối cảnh dễ thương === def ve_boi_canh(): man_hanh.fill(XANH_NHE) # Mây for i in range(6): mx = 90 + i * 160 my = 55 + (i % 2) * 40 pygame.draw.circle(man_hanh, TRANG, (mx, my), 30) pygame.draw.circle(man_hanh, TRANG, (mx+30, my-6), 22) pygame.draw.circle(man_hanh, TRANG, (mx+55, my), 28) # Nền tảng for vt in nen_tang: if vt.y == san_dau: pygame.draw.rect(man_hanh, XANH_DAM, vt) else: pygame.draw.rect(man_hanh, (180,140,100), vt) pygame.draw.rect(man_hanh, XANH_LA, (vt.x, vt.y, vt.w, 5)) # === Tạo màn chơi mới === def tao_man_moi(): global chia_khoa_list, quai_vat_list, so_chia_khoa_da, cua_da_mo, nv_x, nv_y, van_toc_y, dang_dat so_chia_khoa_da = 0 cua_da_mo = False nv_x, nv_y = 60, CHIEU_CAO - nv_cao - 85 van_toc_y = 0 dang_dat = True # Vị trí chìa khóa trên các nền tảng — giống game gốc chia_khoa_list = [ (150, 420), (400, 350), (650, 280) ] # Quái vật đi lại quai_vat_list = [ [280, san_dau-50, 1.8, 180, 520], [650, san_dau-50, -2.2, 480, 850] ] # === Kiểm tra va chạm nền tảng === def kiem_tra_nen(x, y, rong, cao): for vt in nen_tang: if y + cao >= vt.y and y + cao <= vt.y + 14 and x + rong > vt.x and x < vt.x + vt.w and van_toc_y >= 0: return vt.y - cao return None # === Reset toàn game === def reset_game(): global muc_choi muc_choi = 1 tao_man_moi() # === Khởi tạo đầu === tao_man_moi() # === VÒNG LẶP CHÍNH === dang_chay = True while dang_chay: phim = pygame.key.get_pressed() # ========== MÀN HÌNH MỞ ĐẦU ========== if trang_thai == MAN_HINH_MO_DAU: ve_boi_canh() t1 = font_lon.render("🔑 CHIIKAWA TÌM CHÌA KHÓA 🔑", True, (102, 51, 0)) t2 = font_vua.render("Giống game: oibking.itch.io/chiikawa-game", True, (80,60,40)) t3 = font_nho.render("Thu thập đủ 3 chìa khóa → Mở cửa → Tránh quái vật → Qua màn!", True, (60,60,60)) t4 = font_nho.render("Nhấn SPACE để bắt đầu chơi", True, (50,50,50)) man_hanh.blit(t1, (CHIEU_RONG//2 - t1.get_width()//2, 100)) man_hanh.blit(t2, (CHIEU_RONG//2 - t2.get_width()//2, 165)) man_hanh.blit(t3, (CHIEU_RONG//2 - t3.get_width()//2, 230)) man_hanh.blit(t4, (CHIEU_RONG//2 - t4.get_width()//2, 300)) ve_chiikawa(CHIEU_RONG//2 - 28, 360, 1.3) for e in pygame.event.get(): if e.type == pygame.QUIT: dang_chay = False if e.type == pygame.KEYDOWN and e.key == pygame.K_SPACE: trang_thai = SANH_CHO # ========== SẢNH CHỜ ========== elif trang_thai == SANH_CHO: ve_boi_canh() t1 = font_lon.render(f"🏃 MÀN {muc_choi} 🏃", True, (102, 51, 0)) t2 = font_vua.render(f"Chìa khóa: {so_chia_khoa_da}/{so_chia_khoa_can} 🔑", True, (180,140,0)) t3 = font_nho.render("← → Di chuyển | SPACE Nhảy", True, (60,60,60)) t4 = font_nho.render("Bật loa để nghe âm thanh 🔊", True, (0,150,0)) man_hanh.blit(t1, (CHIEU_RONG//2 - t1.get_width()//2, 90)) man_hanh.blit(t2, (CHIEU_RONG//2 - t2.get_width()//2, 160)) man_hanh.blit(t3, (CHIEU_RONG//2 - t3.get_width()//2, 380)) man_hanh.blit(t4, (CHIEU_RONG//2 - t4.get_width()//2, 420)) ve_chiikawa(CHIEU_RONG//2 - 28, 200, 1.4) for e in pygame.event.get(): if e.type == pygame.QUIT: dang_chay = False if e.type == pygame.KEYDOWN and e.key == pygame.K_SPACE: trang_thai = DANG_CHOI # ========== ĐANG CHƠI ========== elif trang_thai == DANG_CHOI: # Di chuyển trái/phải if phim[pygame.K_LEFT]: nv_x -= toc_do_chay if phim[pygame.K_RIGHT]: nv_x += toc_do_chay nv_x = max(0, min(CHIEU_RONG - nv_rong, nv_x)) # Nhảy + Âm thanh if phim[pygame.K_SPACE] and dang_dat: van_toc_y = -toc_do_nhay dang_dat = False tieng_nhay.play() # 🔊 Nghe tiếng nhảy # Vật lý rơi van_toc_y += trong_luc nv_y += van_toc_y mat_nen = kiem_tra_nen(nv_x, nv_y, nv_rong, nv_cao) if mat_nen: nv_y = mat_nen van_toc_y = 0 dang_dat = True # Quái vật đi lại liên tục for q in quai_vat_list: q[0] += q[2] if q[0] < q[3] or q[0] > q[4]: q[2] = -q[2] # Thu thập chìa khóa + Âm thanh for vt in chia_khoa_list[:]: if nv_x < vt[0]+24 and nv_x+nv_rong > vt[0] and nv_y < vt[1]+34 and nv_y+nv_cao > vt[1]: chia_khoa_list.remove(vt) so_chia_khoa_da += 1 tieng_lay_chia_khoa.play() # 🔊 Nghe tiếng lấy chìa khóa # Đủ chìa khóa → Mở cửa + Âm thanh if so_chia_khoa_da >= so_chia_khoa_can and not cua_da_mo: cua_da_mo = True tieng_mo_cua.play() # 🔊 Nghe tiếng mở cửa # Vào cửa → Qua màn + Âm thanh if cua_da_mo and nv_x+nv_rong > cua_ra.x and nv_x < cua_ra.x+cua_ra.w and nv_y+nv_cao > cua_ra.y: muc_choi += 1 tieng_qua_muc.play() # 🔊 Nghe tiếng qua màn tao_man_moi() trang_thai = KE_THUC_MUC # Va chạm quái vật → Thua + Âm thanh hop_nv = pygame.Rect(nv_x+10, nv_y+10, nv_rong-20, nv_cao-20) for q in quai_vat_list: hop_q = pygame.Rect(q[0]+6, q[1]+6, 33, 33) if hop_nv.colliderect(hop_q): tieng_thua.play() # 🔊 Nghe tiếng thua trang_thai = KET_THUC # Vẽ mọi thứ lên màn hình ve_boi_canh() ve_cua(cua_da_mo) for vt in chia_khoa_list: ve_chia_khoa(vt) for q in quai_vat_list: ve_quai(q) ve_chiikawa(nv_x, nv_y, 1.0) # Thông tin trên màn hình t_muc = font_vua.render(f"Màn {muc_choi}", True, (102, 51, 0)) t_khoa = font_vua.render(f"🔑 {so_chia_khoa_da}/{so_chia_khoa_can}", True, (180,140,0)) man_hanh.blit(t_muc, (15, 10)) man_hanh.blit(t_khoa, (15, 50)) if cua_da_mo: t_bao = font_vua.render("✅ CỬA ĐÃ MỞ! Chạy vào bên phải!", True, (0,150,0)) man_hanh.blit(t_bao, (CHIEU_RONG//2 - t_bao.get_width()//2, 10)) for e in pygame.event.get(): if e.type == pygame.QUIT: dang_chay = False # ========== KẾT THÚC MÀN ========== elif trang_thai == KE_THUC_MUC: ve_boi_canh() t1 = font_lon.render(f"🎉 HOÀN THÀNH MÀN {muc_choi-1}! 🎉", True, (0,150,0)) t2 = font_vua.render(f"Đang chuẩn bị màn {muc_choi}...", True, (80,60,40)) t3 = font_nho.render("Nhấn SPACE để tiếp tục phiêu lưu", True, (60,60,60)) man_hanh.blit(t1, (CHIEU_RONG//2 - t1.get_width()//2, 140)) man_hanh.blit(t2, (CHIEU_RONG//2 - t2.get_width()//2, 210)) man_hanh.blit(t3, (CHIEU_RONG//2 - t3.get_width()//2, 290)) ve_chiikawa(CHIEU_RONG//2 - 28, 360, 1.2) for e in pygame.event.get(): if e.type == pygame.QUIT: dang_chay = False if e.type == pygame.KEYDOWN and e.key == pygame.K_SPACE: trang_thai = DANG_CHOI # ========== KẾT THÚC GAME ========== elif trang_thai == KET_THUC: ve_boi_canh() t1 = font_lon.render("💔 Bị quái vật bắt rồi!", True, (204, 0, 0)) t2 = font_vua.render(f"Đã vượt qua: {muc_choi-1} màn chơi", True, (102, 51, 0)) t3 = font_nho.render("SPACE=Chơi lại từ đầu | Q=Thoát game", True, (60,60,60)) man_hanh.blit(t1, (CHIEU_RONG//2 - t1.get_width()//2, 130)) man_hanh.blit(t2, (CHIEU_RONG//2 - t2.get_width()//2, 200)) man_hanh.blit(t3, (CHIEU_RONG//2 - t3.get_width()//2, 290)) ve_chiikawa(CHIEU_RONG//2 - 28, 360, 1.1) for e in pygame.event.get(): if e.type == pygame.QUIT: dang_chay = False if e.type == pygame.KEYDOWN: if e.key == pygame.K_SPACE: reset_game() trang_thai = SANH_CHO if e.key == pygame.K_q: dang_chay = False pygame.display.flip() dong_ho.tick(60) pygame.quit() sys.exit()
